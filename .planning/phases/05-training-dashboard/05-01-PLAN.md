```python
_log = logging.getLogger("dashboard.auth")

def get_password() -> str:
    return os.environ.get("DASHBOARD_PASSWORD", "")

class PasswordAuthMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        if request.url.path.startswith("/api/"):
            password = get_password()
            if password:
                token = request.headers.get("Authorization", "").removeprefix("Bearer ").strip()
                if not hmac.compare_digest(token, password):
                    raise HTTPException(status_code=401, detail="Unauthorized")
        return await call_next(request)

async def verify_ws_token(token: str) -> None:
    password = get_password()
    if password and not hmac.compare_digest(token, password):
        raise WebSocketException(code=status.WS_1008_POLICY_VIOLATION)
```

- `hmac.compare_digest` used for timing-safe password comparison per RESEARCH.md security section
- `get_password()` reads env var every call so it can be changed at runtime
- Auth skipped when `DASHBOARD_PASSWORD` is empty/unset per D-16
